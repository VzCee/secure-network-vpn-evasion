Switch 1 (Acceso y Seguridad de Capa 2)

enable
configure terminal
!
! Nombres de las VLANs corregidos
vlan 30
 name VLAN_USUARIOS_VPN
exit
vlan 40
 name VLAN_RESTRINGIDA
exit
!
! Activamos seguridad global (DHCP Snooping y ARP Inspection)
ip dhcp snooping
ip dhcp snooping vlan 30,40
ip arp inspection vlan 30,40
!
! Puerto Troncal hacia el Router 2 (Confiable)
interface e0/0
 switchport trunk encapsulation dot1q
 switchport mode trunk
 ip dhcp snooping trust
 ip arp inspection trust
exit
!
! Puerto de la VLAN 40 (Restringida / IP .71)
interface e0/1
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
exit
!
! Puerto de la VLAN 30 (Acceso VPN / IP .7)
interface e0/2
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
exit
!
do write

Router 2 (Gateway, ACL y Seguridad de Capa 3)

enable
configure terminal
!
! Desactivamos protocolos inseguros y aseguramos accesos
no cdp run
no vtp mode
aaa new-model
username 2024-2372 secret Itla2372
line vty 0 4
 transport input ssh
 login local
exit
!
! Excluimos los Gateways del DHCP
ip dhcp excluded-address 10.23.72.1
ip dhcp excluded-address 10.23.72.65
!
! Pool de la VLAN 30 (La que corona)
ip dhcp pool VLAN30
 network 10.23.72.0 255.255.255.192
 default-router 10.23.72.1
 dns-server 8.8.8.8
exit
!
! Pool de la VLAN 40 (La bloqueada)
ip dhcp pool VLAN40
 network 10.23.72.64 255.255.255.192
 default-router 10.23.72.65
 dns-server 8.8.8.8
exit
!
! La lista de control de acceso (ACL) que bloquea la VPN a la 40
ip access-list extended ACL_VPN
 10 deny udp 10.23.72.64 0.0.0.63 any eq 1701
 20 permit ip any any
exit
!
! Subinterfaces (El ruteo inter-vlan)
interface e0/1
 no shutdown
 duplex full
exit
!
interface e0/1.30
 encapsulation dot1Q 30
 ip address 10.23.72.1 255.255.255.192
exit
!
interface e0/1.40
 encapsulation dot1Q 40
 ip address 10.23.72.65 255.255.255.192
 ip access-group ACL_VPN in
exit
!
! Puerto hacia el Fortinet y ruta por defecto
interface e0/0
 ip address 172.16.1.1 255.255.255.252
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 172.16.1.2
!
do write

Fortinet (Ruteo, Objetos y Políticas de Evasión)

config router static
    edit 1
        set dst 10.23.72.0 255.255.255.128
        set gateway 172.16.1.1
        set device "port1"
    next
    edit 2
        set dst 172.16.3.0 255.255.255.252
        set gateway 172.16.3.2
        set device "port2"
    next
    edit 3
        set dst 192.23.72.0 255.255.255.0
        set gateway 172.16.3.2
        set device "port2"
    next
end

config firewall address
    edit "RED_VLANS"
        set subnet 10.23.72.0 255.255.255.128
    next
    edit "SERVIDOR_WEB"
        set subnet 10.23.72.130 255.255.255.255
    next
    edit "RED_VPN"
        set subnet 192.23.72.0 255.255.255.0
    next
end

config firewall policy
    edit 1
        set name "Bloqueo_HTTP"
        set srcintf "port1"
        set dstintf "port2"
        set srcaddr "RED_VLANS"
        set dstaddr "SERVIDOR_WEB"
        set action deny
        set schedule "always"
        set service "HTTP"
    next
    edit 2
        set name "Permitir_VPN"
        set srcintf "port2"
        set dstintf "port2"
        set srcaddr "RED_VPN"
        set dstaddr "SERVIDOR_WEB"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
end


Router 3 (Servidor VPN LNS)

enable
configure terminal
!
! Seguridad básica
no cdp run
aaa new-model
username admin secret admin
line vty 0 4
 transport input ssh
 login local
exit
!
! Configuración de la VPN L2TP
ip local pool VPN_POOL 192.23.72.10 192.23.72.50
username 2024-2372 password Itla2372
!
vpdn enable
vpdn-group L2TP_GROUP
 accept-dialin
  protocol l2tp
  virtual-template 1
 terminate-from hostname R3
exit
!
interface Virtual-Template1
 ip unnumbered e0/0
 peer default ip address pool VPN_POOL
 ppp authentication chap ms-chap ms-chap-v2
exit
!
! Interfaces y ruteo
interface e0/0
 ip address 172.16.3.2 255.255.255.252
 no shutdown
exit
!
interface e0/1
 ip address 10.23.72.129 255.255.255.192
 no shutdown
exit
!
ip route 0.0.0.0 0.0.0.0 172.16.3.1
ip route 10.23.72.0 255.255.255.128 172.16.3.1
!
do write
